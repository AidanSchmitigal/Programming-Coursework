int main() {
  int size = 5;
  int* array = createArray(size);
  int sizes = sizeof(array) / sizeof(int);
  delete[] array;
  if (sizes == 5) return 0;
  std::cout << "Your array is incorrect. Your array was " << sizes << " long.";

}
