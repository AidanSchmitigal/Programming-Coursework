Write a function to return if a number is positive.
--- Displayed Header and Footer Code ---
// Function Header
bool isPositive(int num);

int main() {
  if (isPositive(5)) {
    cout << "5 is positive" << endl;
  } else {
    cout << "5 is not positive" << endl;
  }
  if (isPositive(-5)) {
    cout << "-5 is positive" << endl;
  } else {
    cout << "-5 is not positive" << endl;
  }
  return 0;
}
